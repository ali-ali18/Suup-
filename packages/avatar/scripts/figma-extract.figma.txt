// (arquivo .txt para o lint não tratar como módulo: o `return` final é do corpo da função do `use_figma`)
// Código que roda DENTRO do Figma (ferramenta `use_figma` do MCP da Figma, arquivo
// c1FraH70ehfVHADeOXKspp). Não roda no Node. Lê os 17 conjuntos de componentes da página
// "Mascotes" e devolve, por personagem, a grade 19x19 do estado "Parado" e, para cada outro estado,
// só as células que mudam. O resultado, junto com os quadros da página "Animação — folha" (5:2),
// vira `figma-pixel.json`; `gen-pixel-data.mjs` transforma esse JSON nos módulos TypeScript.
//
// Descoberta (nos dados, não por palpite): todo retângulo tem 12 x 12, em x/y = 4 + 12k, dentro do
// quadro de 236 x 236 (4 + 19 x 12 + 4). A unidade do pixel é 12 e a grade é 19 x 19.
await figma.setCurrentPageAsync(figma.root.children[0]);
const f = await figma.getNodeByIdAsync("1:2");
const sets = f.findAllWithCriteria({ types: ["COMPONENT_SET"] });
const hex = (c) =>
  [c.r, c.g, c.b].map((v) => Math.round(v * 255).toString(16).padStart(2, "0")).join("");
const out = [];
for (const s of sets) {
  const grids = {};
  let body = null;
  for (const v of s.children) {
    const g = Array.from({ length: 19 }, () => Array(19).fill("."));
    for (const c of v.children) {
      const cx = (c.x - 4) / 12;
      const cy = (c.y - 4) / 12;
      let ch;
      if (c.name === "corpo") {
        body = hex(c.fills[0].color);
        ch = "B";
      } else ch = { olho: "e", acessório: "a", lâmpada: "l" }[c.name];
      g[cy][cx] = ch;
    }
    grids[v.name.replace("Estado=", "")] = g.map((r) => r.join(""));
  }
  const base = grids.Parado;
  const d = {};
  for (const k in grids) {
    if (k === "Parado") continue;
    const l = [];
    for (let y = 0; y < 19; y++)
      for (let x = 0; x < 19; x++)
        if (grids[k][y][x] !== base[y][x]) l.push(x.toString(36) + y.toString(36) + grids[k][y][x]);
    d[k] = l.join(" ");
  }
  out.push({ id: s.name.replace("Mascote/", ""), body, base: base.join("/"), d });
}
return JSON.stringify(out);
